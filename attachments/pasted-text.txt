  const ConfluenceEngine = {
    analyze(assetName = State.currentChartAsset) {
      const buf = RealCandleTracker.getBuffer(assetName);
      const history = buf.history;
      const asset = assetName;

      const triggerEvacuate = (reason) => {
        setSignal('NEUTRO', 30, 50, 50, `⚠️ ${reason}`);
        return { canTrade: false, evacuate: true, reason };
      };

      const blockWait = (label, conf = 0) => {
        setSignal('NEUTRO', conf, 50, 50, label);
        return { canTrade: false, evacuate: false, reason: label };
      };

      const fire = (dir, conf, scenarioName, reason) => {
        const isCall = dir === 'COMPRA';
        setSignal(dir, conf, isCall ? conf : 100 - conf, isCall ? 100 - conf : conf, scenarioName);
        return { canTrade: true, evacuate: false, direction: dir, confidence: conf, reason };
      };

      if (!isOTC(asset)) return triggerEvacuate('EXIGE PAR OTC');
      if (BLACKLIST_EMPATE_KEYS.has(normKey(asset))) return triggerEvacuate('PAR COM HISTÓRICO DE EMPATES');

      if (history.length < MIN_HISTORY) {
        return blockWait(`LENDO CONTEXTO (${history.length}/${MIN_HISTORY} VELAS)...`, 35);
      }

      const last = history[history.length - 1]; // Vela anterior imediata
      const prev = history[history.length - 2]; // Penúltima vela

      const range = last.high - last.low;
      if (range <= 0) return blockWait('VELA SEM RANGE (DOJI LINHA)', 20);

      const body = Math.abs(last.close - last.open);
      const bodyRatio = body / range;
      const upperWick = last.high - Math.max(last.open, last.close);
      const lowerWick = Math.min(last.open, last.close) - last.low;

      // 1. FILTRO DE CORPO ANÊMICO / INDECISÃO (Elimina perdas como AED/CNY no vídeo)
      if (bodyRatio < 0.52) {
        return blockWait(`CORPO FRACO (${Math.round(bodyRatio * 100)}% < 52%) - INDECISÃO`, 25);
      }

      // 2. CONTAGEM DE VELAS CONSECUTIVAS (FILTRO DE EXAUSTÃO - Elimina perdas em NZD/USD e EUR/USD)
      let consecutiveRed = 0;
      let consecutiveGreen = 0;

      for (let i = history.length - 1; i >= 0; i--) {
        if (history[i].isRed) {
          if (consecutiveGreen > 0) break;
          consecutiveRed++;
        } else if (history[i].isGreen) {
          if (consecutiveRed > 0) break;
          consecutiveGreen++;
        } else break;
      }

      // Se já temos 3 ou mais velas da mesma cor, o movimento está esticado (alto risco de reversão)
      if (last.isRed && consecutiveRed >= 3) {
        return blockWait(`EXAUSTÃO DE BAIXA: ${consecutiveRed} VELAS VERMELHAS SEGUIDAS`, 30);
      }
      if (last.isGreen && consecutiveGreen >= 3) {
        return blockWait(`EXAUSTÃO DE ALTA: ${consecutiveGreen} VELAS VERDES SEGUIDAS`, 30);
      }

      // 3. FILTRO DE PAVIO DE REJEIÇÃO
      // Se for vela vermelha, mas deixou pavio inferior grande (> 30%), compradores estão defendendo o fundo
      if (last.isRed && (lowerWick / range) > 0.30) {
        return blockWait(`REJEIÇÃO DE FUNDO: PAVIO INFERIOR ${(lowerWick / range * 100).toFixed(0)}%`, 35);
      }

      // Se for vela verde, mas deixou pavio superior grande (> 30%), vendedores estão defendendo o topo
      if (last.isGreen && (upperWick / range) > 0.30) {
        return blockWait(`REJEIÇÃO DE TOPO: PAVIO SUPERIOR ${(upperWick / range * 100).toFixed(0)}%`, 35);
      }

      // 4. FILTRO DE DESACELERAÇÃO (Perda de Momentum contra a vela anterior)
      const prevBody = Math.abs(prev.close - prev.open);
      if (prevBody > 0 && body < prevBody * 0.50 && last.isRed === prev.isRed) {
        return blockWait('DESACELERAÇÃO DE MOMENTUM (CORPO REDUZIU > 50%)', 30);
      }

      // ENTRADA AUTORIZADA
      if (last.isGreen) {
        return fire('COMPRA', 92, `🟢 FLUXO DE ALTA CONFIRMADO (${State.expirationSeconds}s)`, 'Corpo forte sem rejeição superior');
      } else if (last.isRed) {
        return fire('VENDA', 92, `🔴 FLUXO DE BAIXA CONFIRMADO (${State.expirationSeconds}s)`, 'Corpo forte sem rejeição inferior');
      } else {
        return blockWait('DOJI PURO', 20);
      }
    }
  };

  function auditTradeResultAccurate(initialBalance, stakeUsed, payoutPct, callback) {
    const deadline = Date.now() + AUDIT_TIMEOUT_MS;
    let settled = false;
    const expectedProfit = Number((stakeUsed * (payoutPct / 100)).toFixed(2));

    const iv = setInterval(() => {
      if (settled) return;
      const currentBal = BrokerDOM.readBalance();
      const netDelta = Number((currentBal - initialBalance).toFixed(2));

      if (netDelta >= expectedProfit * 0.4) {
        settled = true;
        clearInterval(iv);
        callback('WIN', netDelta > 0 ? netDelta : expectedProfit);
        return;
      }

      if (Math.abs(netDelta) <= 0.05 && (Date.now() > deadline - (AUDIT_TIMEOUT_MS - 2000))) {
        settled = true;
        clearInterval(iv);
        callback('EMPATE', 0);
        return;
      }

      if (Date.now() >= deadline) {
        settled = true;
        clearInterval(iv);
        if (netDelta >= expectedProfit * 0.3) {
          callback('WIN', netDelta);
        } else if (Math.abs(netDelta) <= 0.1) {
          callback('EMPATE', 0);
        } else {
          callback('LOSS', -Number(stakeUsed.toFixed(2)));
        }
      }
    }, 150);
  }

  function handleTradeResult(status, delta, activePair) {
    console.log(`[ORION AUDIT] Resultado: ${status} | Delta: ${delta >= 0 ? '+' : ''}${delta} | Ativo: ${activePair}`);

    if (status === 'WIN') {
      State.wins++;
      State.consecutiveLosses = 0;
      State.currentProfit = Number((State.currentProfit + delta).toFixed(2));

      if (State.managementMode === 'SOROS') {
        if (State.currentSorosLevel >= State.sorosMaxLevel) {
          State.tradeStatusText = `🎉 SOROS BATIDO (+${delta})! Voltando Nível 1...`;
          State.currentSorosLevel = 1;
          State.currentStake = State.entryAmount;
        } else {
          State.currentSorosLevel++;
          State.currentStake = Number((State.currentStake + delta).toFixed(2));
          State.tradeStatusText = `✅ WIN! Soros Nível ${State.currentSorosLevel} -> $${State.currentStake.toFixed(2)}`;
        }
      } else if (State.managementMode === 'SOROS_CONDICIONAL') {
        State.condicionalWinsCount++;
        if (State.condicionalWinsCount >= 2) {
          State.currentSorosLevel = 2;
          State.currentStake = Number((State.entryAmount + delta).toFixed(2));
          State.tradeStatusText = `🔥 2 WINS SEGUIDOS! Soros N2: $${State.currentStake.toFixed(2)}`;
        } else {
          State.currentStake = State.entryAmount;
          State.tradeStatusText = `✅ WIN 1/2 Mão Fixa! Mais 1 p/ Soros.`;
        }
      } else {
        State.currentStake = State.entryAmount;
        State.tradeStatusText = `✅ WIN +$${delta.toFixed(2)} Mão Fixa!`;
      }
    } else if (status === 'EMPATE') {
      State.empates++;
      State.tradeStatusText = '⚪ EMPATE! Mantendo stake.';
    } else {
      State.losses++;
      State.consecutiveLosses++;
      State.currentProfit = Number((State.currentProfit + delta).toFixed(2));
      State.currentSorosLevel = 1;
      State.condicionalWinsCount = 0;
      State.currentStake = State.entryAmount;
      State.tradeStatusText = `❌ LOSS -$${Math.abs(delta).toFixed(2)}! Resetando p/ $${State.entryAmount.toFixed(2)}`;
    }

    if (State.stopLossLimit > 0 && State.consecutiveLosses >= State.stopLossLimit) {
      State.isRunning = false;
      State.tradeStatusText = `🛑 STOP LOSS ATINGIDO (${State.consecutiveLosses} derrotas)!`;
      updatePowerButtonUI();
      State.tradeInProgress = false;
      State.orderLock = false;
      renderHud();
      return;
    }

    if (State.currentProfit >= State.targetMeta) {
      State.isRunning = false;
      State.tradeStatusText = `🏆 META BATIDA (+$${State.currentProfit.toFixed(2)})!`;
      updatePowerButtonUI();
      State.tradeInProgress = false;
      State.orderLock = false;
      renderHud();
      return;
    }

    renderHud();
    State.lastTradedAssetName = activePair;
    pushRecent(normKey(activePair));

    setTimeout(() => {
      State.tradeStatusText = 'Trocando de ativo para diversificar...';
      renderHud();
      const nextVIP = BrokerDOM.pickNextVIPAsset();
      BrokerDOM.typeAndSwitchAsset(nextVIP, () => {
        State.tradeInProgress = false;
        State.orderLock = false;
        renderHud();
      });
    }, 450);
  }

  function executeEntryNow(decision, activePair, balanceBefore) {
    const stake = State.currentStake;
    const payout = State.currentChartPayout || 92;
    State.lastEntryStake = stake;

    if (!BrokerDOM.setStake(stake)) {
      State.tradeStatusText = 'Campo de stake não localizado!';
      State.tradeInProgress = false;
      State.waitingForZeroSeconds = false;
      State.orderLock = false;
      renderHud();
      return;
    }

    State.tradeStatusText = `🎯 DISPARO (${State.expirationSeconds}s): [${decision.direction}] $${stake.toFixed(2)}`;
    renderHud();

    const clickSuccess = BrokerDOM.clickTrade(decision.direction);

    if (!clickSuccess) {
      State.tradeStatusText = 'Falha ao acionar botão!';
      State.tradeInProgress = false;
      State.waitingForZeroSeconds = false;
      State.orderLock = false;
      renderHud();
      return;
    }

    State.waitingForZeroSeconds = false;
    const waitMs = Math.max(1000, (State.expirationSeconds * 1000) + 400);

    setTimeout(() => {
      State.tradeStatusText = 'Auditando resultado real...';
      renderHud();

      auditTradeResultAccurate(balanceBefore, stake, payout, (status, delta) => {
        handleTradeResult(status, delta, activePair);
      });
    }, waitMs);
  }

  function scheduleEngineCycle() {
    if (!State.isRunning || State.tradeInProgress || State.isTypingAsset || State.orderLock) return;

    const now = Date.now();
    const sync = getCandleSync(State.expirationSeconds);
    const activePair = State.currentChartAsset;
    const activeKey = normKey(activePair);
    const cooldownOk = now - State.lastSwitchTimestamp > SWITCH_COOLDOWN_MS;

    if (activeKey === normKey(State.lastTradedAssetName) || State.recentTradedQueue.includes(activeKey)) {
      if (cooldownOk) BrokerDOM.typeAndSwitchAsset(BrokerDOM.pickNextVIPAsset());
      return;
    }

    BrokerDOM.readCurrentPayout();
    if (State.payoutKnown && State.currentChartPayout < State.minPayoutFilter) {
      if (cooldownOk) {
        State.tradeStatusText = `Payout ${State.currentChartPayout}% < ${State.minPayoutFilter}% - Trocando...`;
        BrokerDOM.typeAndSwitchAsset(BrokerDOM.pickNextVIPAsset());
      }
      return;
    }

    const decision = ConfluenceEngine.analyze();

    if (decision.evacuate) {
      if (cooldownOk) {
        State.tradeStatusText = `🛡️ ${decision.reason} -> Trocando par...`;
        renderHud();
        BrokerDOM.typeAndSwitchAsset(BrokerDOM.pickNextVIPAsset());
      }
      return;
    }

    if (!decision.canTrade) {
      const age = now - State.lastSwitchTimestamp;
      const buf = RealCandleTracker.getBuffer(activePair);
      if (buf.history.length >= MIN_HISTORY && age > SCAN_TIMEOUT_MS && cooldownOk) {
        State.tradeStatusText = `Sem sinal em ${activePair}. Trocando...`;
        BrokerDOM.typeAndSwitchAsset(BrokerDOM.pickNextVIPAsset());
      }
      return;
    }

    if (!State.waitingForZeroSeconds) {
      const balanceBefore = BrokerDOM.readBalance();
      if (balanceBefore <= 0 || State.currentStake > balanceBefore) {
        State.tradeStatusText = 'Saldo insuficiente!';
        renderHud();
        return;
      }

      State.waitingForZeroSeconds = true;
      State.pendingEntry = { decision, activePair, balanceBefore };
    }

    if (State.waitingForZeroSeconds && State.pendingEntry) {
      State.tradeStatusText = `⏳ ARMADO: ${decision.direction} (${decision.reason}) • Virada (${sync.secondsUntilNextCandle}s)`;
      renderHud();

      if (sync.isAtZeroThreshold && State.lastTradedCandleId !== sync.candleId && !State.orderLock) {
        State.tradeInProgress = true;
        State.lastTradedCandleId = sync.candleId;
        const entry = State.pendingEntry;
        State.pendingEntry = null;
        executeEntryNow(entry.decision, entry.activePair, entry.balanceBefore);
      }
    }
  }
